# Motion across the site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every page of the Hu/man site "B · Alive" motion (entrance reveals, gentle ambient loops, a services ticker, living shader gradients, an RGB navbar logo) without dropping below Lighthouse 95 or WCAG 2.2 AA.

**Architecture:** CSS does the cheap work (default reveals, variants, ambient loops, hovers, the logo hue cycle, collage assembly). A small boot script (`src/scripts/motion/index.ts`) keeps the existing IntersectionObserver for reveals and, once the page is idle, lazy-loads three pinned libraries: GSAP + SplitText (heading line reveals), Lenis (smooth wheel scrolling) and Paper Shaders (WebGL mesh gradients on at most 3 panels per page). Everything is off under reduced motion, and all content is visible without JavaScript.

**Tech Stack:** Astro 7, TypeScript, plain CSS, `gsap@3.15.0`, `lenis@1.3.26`, `@paper-design/shaders@0.0.81`, Playwright (checks), Node's test runner.

**Spec:** `docs/superpowers/specs/2026-10-27-motion-design.md`

**Ground rules for every task (from CLAUDE.md and the spec):**
- Work in `site/`. Run commands from `site/`.
- Never animate the hero `h1` (`#hero-title`) or any `InnerHero` `h1` (`#ihero-title`): they are the LCP element.
- Animate only `opacity` and the individual transform properties `translate`, `rotate`, `scale` (they compose with any existing `transform`, so centring rules like `translateX(-50%)` keep working). Never animate `clip-path` or filters.
- Every looping or decorative animation goes inside `@media (prefers-reduced-motion: no-preference)`. The global reduced-motion rule at the end of `src/styles/base.css` stays.
- No third-party requests: libraries are bundled from npm, versions pinned exactly (no `^`).

```ts
/**
 * Colours for the moving shader gradients, drawn from the palette in src/styles/tokens.css. Text
 * sits on two panels, so every colour there keeps that text at WCAG AA (tests/palettes.test.mjs);
 * that is why the indigo and deep orange of the static --grad-talk are left out of `talk`.
 */
export type ShaderKey = 'ideas' | 'talk' | 'arch';

export interface ShaderPalette {
  colors: string[];
  /** The text colour drawn over the panel, or null when there is no text on it. */
  text: string | null;
  /** Paper Shaders speed: 0.15-0.4 keeps a background calm. */
  speed: number;
}

export const palettes: Record<ShaderKey, ShaderPalette> = {
  ideas: { colors: ['#2a3875', '#4c29dd', '#5c48dd', '#3b2fb4'], text: '#f1f2ec', speed: 0.2 },
  talk: { colors: ['#f8cea2', '#e79569', '#e85e8f', '#d9545a'], text: '#0b0d0a', speed: 0.25 },
  arch: { colors: ['#2c4377', '#3f7fa3', '#4a9bb0', '#edd1a8'], text: null, speed: 0.2 },
};

const channel = (hex: string, i: number): number => {
  const c = parseInt(hex.slice(1 + i * 2, 3 + i * 2), 16) / 255;
  return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
};

const luminance = (hex: string): number =>
  0.2126 * channel(hex, 0) + 0.7152 * channel(hex, 1) + 0.0722 * channel(hex, 2);

/** WCAG 2 contrast ratio of two #rrggbb colours. */
export function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}
```

- [ ] **Step 1: Write the check script**

`site/scripts/motion-check.mjs`:
```js
/**
 * Behaviour checks for the site's motion, against the built dist/ (run `npm run build` first):
 * - normal motion: the navbar logo's hue changes over time, the ticker runs and its pause button
 *   toggles, shaders mount (at most 3 per page), no console errors;
 * - reduced motion: no running animations, no shader canvases, no Lenis, every reveal visible.
 * Exits 1 on any failure.
 */
import { chromium } from 'playwright';
import { startServer } from './serve.mjs';

const PORT = 4393;
const PAGES = ['', 'services/', 'pricing/', 'about/', 'contact/', 'terms/'];
const server = await startServer(new URL('../dist/', import.meta.url).pathname, PORT);
const browser = await chromium.launch({ args: ['--use-gl=angle', '--use-angle=swiftshader'] });
const failures = [];
const fail = (where, what) => failures.push(`${where}: ${what}`);

for (const path of PAGES) {
  const url = `http://localhost:${PORT}/${path}`;

  // Normal motion.
  const page = await browser.newPage({ viewport: { width: 1280, height: 900 } });
  const errors = [];
  page.on('console', (m) => m.type() === 'error' && errors.push(m.text()));
  page.on('pageerror', (e) => errors.push(e.message));
  await page.goto(url, { waitUntil: 'load' });
  await page.evaluate(async () => {
    for (let y = 0; y < document.body.scrollHeight; y += 600) {
      window.scrollTo(0, y);
      await new Promise((r) => setTimeout(r, 120));
    }
    window.scrollTo(0, 0);
  });
  await page.waitForTimeout(1500);
  const logoColour = () =>
    page.evaluate(() => {
      const path = document.querySelector('.site-header__logo .logo path');
      return path ? getComputedStyle(path).fill : '';
    });
  const c1 = await logoColour();
  await page.waitForTimeout(1500);
  const c2 = await logoColour();
  if (!c1 || c1 === c2) fail(path || '/', `logo colour did not change (${c1} -> ${c2})`);
  const canvases = await page.locator('[data-shader] canvas').count();
  if (canvases > 3) fail(path || '/', `${canvases} shader canvases (max 3)`);
  if (path === '' && canvases < 1) fail('/', 'no shader mounted on the home page');
  if (path === '') {
    const toggle = page.locator('.ticker__toggle');
    if ((await toggle.count()) !== 1) fail('/', 'ticker pause button missing');
    else {
      await toggle.click();
      if ((await toggle.getAttribute('aria-pressed')) !== 'true') fail('/', 'ticker did not pause');
    }
  }
  if (errors.length) fail(path || '/', `console errors: ${errors.join(' | ')}`);
  await page.close();

  // Reduced motion.
  const calm = await browser.newPage({
    viewport: { width: 1280, height: 900 },
    reducedMotion: 'reduce',
  });
  await calm.goto(url, { waitUntil: 'load' });
  await calm.waitForTimeout(1200);
  const state = await calm.evaluate(() => ({
    running: document
      .getAnimations()
      .filter((a) => a.playState === 'running' && a.effect?.getTiming().iterations === Infinity)
      .length,
    canvases: document.querySelectorAll('[data-shader] canvas').length,
    lenis: document.documentElement.classList.contains('lenis'),
    hidden: [...document.querySelectorAll('[data-reveal]')].filter(
      (el) => getComputedStyle(el).opacity === '0',
    ).length,
  }));
  if (state.running) fail(path || '/', `reduced motion: ${state.running} looping animations`);
  if (state.canvases) fail(path || '/', 'reduced motion: shader canvases mounted');
  if (state.lenis) fail(path || '/', 'reduced motion: Lenis is on');
  if (state.hidden) fail(path || '/', `reduced motion: ${state.hidden} reveals still hidden`);
  await calm.close();
}

await browser.close();
server.close();
if (failures.length) {
  console.error(`motion-check: ${failures.length} problem(s)\n  - ${failures.join('\n  - ')}`);
  process.exit(1);
}
console.log(`motion-check: ${PAGES.length} pages, all clear.`);
```

```ts
/**
 * Scroll reveals: each [data-reveal] element plays once as it comes into view (never scrubbed by
 * scrolling) by gaining the `is-in` class. CSS does the default fade-up and the `slide`, `grow`
 * and `scan` variants. `lines` headings are split into lines with GSAP SplitText and slide up
 * from a mask, once GSAP has loaded; until then (or if it never loads) they fade like the rest.
 *
 * Content is visible without JS (the `.js` class gates the hidden state), and reduced motion shows
 * everything at once.
 */
type Gsap = typeof import('gsap').gsap;
type SplitTextClass = typeof import('gsap/SplitText').SplitText;

let gsapLib: { gsap: Gsap; SplitText: SplitTextClass } | null = null;

/** Called by index.ts once GSAP has loaded. */
export function useGsap(gsap: Gsap, SplitText: SplitTextClass): void {
  gsap.registerPlugin(SplitText);
  gsapLib = { gsap, SplitText };
}

function playLines(el: HTMLElement): void {
  if (!gsapLib) return;
  const { gsap, SplitText } = gsapLib;
  SplitText.create(el, {
    type: 'lines',
    mask: 'lines',
    autoSplit: true,
    onSplit: (self) =>
      gsap.from(self.lines, { yPercent: 105, duration: 0.9, stagger: 0.09, ease: 'power3.out' }),
  });
}

export function startReveals(reduceMotion: boolean): void {
  const items = document.querySelectorAll<HTMLElement>('[data-reveal]');
  if (reduceMotion || !('IntersectionObserver' in window)) {
    items.forEach((el) => el.classList.add('is-in'));
    return;
  }
  const io = new IntersectionObserver(
    (entries) => {
      for (const entry of entries) {
        if (!entry.isIntersecting) continue;
        const el = entry.target as HTMLElement;
        if (el.dataset.reveal === 'lines') playLines(el);
        el.classList.add('is-in');
        io.unobserve(el);
      }
    },
    { rootMargin: '0px 0px -8% 0px', threshold: 0.08 },
  );
  items.forEach((el) => io.observe(el));
}
```

- [ ] **Step 2: Create `index.ts`**

`site/src/scripts/motion/index.ts`:
```ts
/**
 * Boots the site's motion. Reveals start straight away (they are cheap). The libraries load when
 * the page is idle, after first paint, so they never delay the LCP: GSAP for heading line
 * reveals, Lenis for smooth wheel scrolling, Paper Shaders for the moving gradients. Under
 * reduced motion none of them load.
 */
import { startReveals, useGsap } from './reveals';

const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

startReveals(reduceMotion);

const whenIdle = (fn: () => void): void => {
  if ('requestIdleCallback' in window) window.requestIdleCallback(fn, { timeout: 2000 });
  else setTimeout(fn, 600);
};

if (!reduceMotion) {
  whenIdle(async () => {
    const [{ gsap }, { SplitText }] = await Promise.all([
      import('gsap'),
      import('gsap/SplitText'),
    ]);
    await document.fonts.ready;
    useGsap(gsap, SplitText);
    const { startSmoothScroll } = await import('./smooth');
    startSmoothScroll();
    const { startShaders } = await import('./shaders');
    startShaders();
  });
}
```

### Task 7: Shader gradients on up to three panels

**Files:** Replace the stub `site/src/scripts/motion/shaders.ts`; modify `IdeasToLife.astro`, `TalkBanner.astro`, `Collage.astro`.

- [ ] **Step 1: Write `shaders.ts`**

```ts
/**
 * Moving mesh gradients (Paper Shaders) behind up to three panels per page, each marked with
 * <span class="shader-layer" data-shader="ideas|talk|arch" aria-hidden="true">. The panel's own
 * CSS gradient shows until the canvas is ready and stays when shaders are skipped: no WebGL,
 * Data Saver on, or a low-memory device. ShaderMount pauses itself off screen; we also pause it
 * while the tab is hidden. Colours: palettes.ts (contrast-tested).
 */
import {
  ShaderMount,
  ShaderFitOptions,
  getShaderColorFromString,
  meshGradientFragmentShader,
} from '@paper-design/shaders';
import { palettes, type ShaderKey } from './palettes';

const MAX_PER_PAGE = 3;

function canRun(): boolean {
  const nav = navigator as Navigator & {
    connection?: { saveData?: boolean };
    deviceMemory?: number;
  };
  if (nav.connection?.saveData) return false;
  if (nav.deviceMemory !== undefined && nav.deviceMemory <= 2) return false;
  try {
    return !!document.createElement('canvas').getContext('webgl2');
  } catch {
    return false;
  }
}

const sizing = {
  u_fit: ShaderFitOptions.cover,
  u_rotation: 0,
  u_scale: 1,
  u_offsetX: 0,
  u_offsetY: 0,
  u_originX: 0.5,
  u_originY: 0.5,
  u_worldWidth: 0,
  u_worldHeight: 0,
};

export function startShaders(): void {
  if (!canRun()) return;
  const layers = [...document.querySelectorAll<HTMLElement>('[data-shader]')].slice(0, MAX_PER_PAGE);
  const mounts: { mount: ShaderMount; speed: number }[] = [];
  for (const layer of layers) {
    const palette = palettes[layer.dataset.shader as ShaderKey];
    if (!palette) continue;
    const colors = palette.colors.map(getShaderColorFromString);
    try {
      const mount = new ShaderMount(
        layer,
        meshGradientFragmentShader,
        {
          ...sizing,
          u_colors: colors,
          u_colorsCount: colors.length,
          u_distortion: 0.8,
          u_swirl: 0.3,
          u_grainMixer: 0,
          u_grainOverlay: 0,
        },
        undefined,
        palette.speed,
        0,
        1, // minPixelRatio: don't upscale a soft gradient
        1920 * 1080, // maxPixelCount: caps work on large/high-DPR screens
      );
      mounts.push({ mount, speed: palette.speed });
      requestAnimationFrame(() => layer.classList.add('is-live'));
    } catch {
      // Leave the CSS gradient in place.
    }
  }
  document.addEventListener('visibilitychange', () => {
    for (const { mount, speed } of mounts) mount.setSpeed(document.hidden ? 0 : speed);
  });
}
```
